import logging as _logging

_resolver_logger = _logging.getLogger(__name__)

def resolve_default_model_path(strict: bool = False) -> Path:
    """Return the first existing default-model path in a known fallback chain.

    Order: DEFAULT_MODEL_PATH → models/retrained/har_model_latest.keras → PRETRAINED_MODEL.
    Raises FileNotFoundError (with all tried paths) if none exist.
    If strict=True, skips the PRETRAINED_MODEL fallback (BUG-3: hand_tapping bias).
    """
    candidates = [
        DEFAULT_MODEL_PATH,
        MODELS_DIR / "retrained" / "har_model_latest.keras",
    ]
    if not strict:
        candidates.append(PRETRAINED_MODEL)

    for p in candidates:
        if p.exists():
            if p != DEFAULT_MODEL_PATH:
                _resolver_logger.warning(
                    "DEFAULT_MODEL_PATH missing (%s); falling back to %s",
                    DEFAULT_MODEL_PATH, p,
                )
            return p

    tried = "\n  - ".join(str(p) for p in candidates)
    raise FileNotFoundError(
        "No default model available. Tried:\n  - "
        + tried
        + "\nRemediation: run dvc pull models/, verify Git LFS, or pass --model <path>."
    )

import pytest

def test_resolve_returns_pinned_default_when_present(tmp_path, monkeypatch):
    pinned = tmp_path / "models" / "retrained" / "har_model_20260327_223917.keras"

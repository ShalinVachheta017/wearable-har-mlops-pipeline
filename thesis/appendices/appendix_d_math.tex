\chapter{Mathematical Definitions}
\label{app:math}

This appendix collects the formulas used in the monitoring, governance, and evaluation chapters in one place. The notation follows Chapters~\ref{chap:monitoring} and~\ref{chap:experiments}: a session produces \(N\) windows, the model outputs a softmax vector \(\mathbf{p}_i=(p_{i,1},\dots,p_{i,K})\) over \(K=11\) classes for window \(i\), the predicted class is \(\hat{y}_i=\arg\max_k p_{i,k}\), and the per-window confidence is \(c_i=\max_k p_{i,k}\).

\section*{D.1 Recognition metrics}

\noindent Accuracy, per-class precision and recall, per-class F1, and macro-averaged F1:
\[
\mathrm{Acc}=\frac{1}{N}\sum_{i=1}^{N}\mathbf{1}\!\left[\hat{y}_i=y_i\right],
\qquad
P_k=\frac{TP_k}{TP_k+FP_k},
\qquad
R_k=\frac{TP_k}{TP_k+FN_k},
\]
\[
F1_k=\frac{2P_kR_k}{P_k+R_k},
\qquad
\mathrm{MacroF1}=\frac{1}{K}\sum_{k=1}^{K}F1_k.
\]

\section*{D.2 Calibration}

\noindent Expected calibration error over \(M\) confidence bins \(B_m\), and per-window predictive entropy:
\[
\mathrm{ECE}=\sum_{m=1}^{M}\frac{|B_m|}{N}\left|\mathrm{acc}(B_m)-\mathrm{conf}(B_m)\right|,
\qquad
H_i=-\sum_{k=1}^{K}p_{i,k}\log p_{i,k}.
\]

\section*{D.3 Monitoring signals}

\noindent Mean confidence, uncertain-window rate (with uncertainty level \(\tau_u\)), temporal transition rate, and dominant-class share (with class counts \(n_k=\sum_i\mathbf{1}[\hat{y}_i=k]\)):
\[
\bar{c}=\frac{1}{N}\sum_{i=1}^{N}c_i,
\qquad
U=\frac{1}{N}\sum_{i=1}^{N}\mathbf{1}\!\left[c_i<\tau_u\right],
\]
\[
T=\frac{1}{N-1}\sum_{i=2}^{N}\mathbf{1}\!\left[\hat{y}_i\neq\hat{y}_{i-1}\right],
\qquad
D=\frac{1}{N}\max_{k}n_k.
\]

\section*{D.4 Distribution drift}

\noindent Per-channel z-score drift against a baseline with mean \(\mu_j\) and standard deviation \(\sigma_j\), and the kernel-based Maximum Mean Discrepancy used in the offline audit branch:
\[
z=\max_{j}\frac{\lvert\hat{\mu}_j-\mu_j\rvert}{\sigma_j},
\]
\[
\mathrm{MMD}^2(X,Y)=\frac{1}{n^2}\sum_{i,i'}k(x_i,x_{i'})+\frac{1}{m^2}\sum_{j,j'}k(y_j,y_{j'})-\frac{2}{nm}\sum_{i,j}k(x_i,y_j).
\]

\section*{D.5 Preprocessing}

\noindent Train-only per-channel standardization with mean \(\mu_j\) and standard deviation \(\sigma_j\) fitted on the Garmin training data:
\[
x'_{i,j}=\frac{x_{i,j}-\mu_j}{\sigma_j},\qquad j\in\{Ax,Ay,Az,Gx,Gy,Gz\}.
\]
The deployed runtime contract is the tuple \(\mathcal{C}=(M,S,L,B,\Omega)\), where \(M\) is the frozen model, \(S\) the train-only scaler, \(L\) the fixed label order, \(B\) the monitoring baseline, and \(\Omega\) the preprocessing and windowing metadata.
